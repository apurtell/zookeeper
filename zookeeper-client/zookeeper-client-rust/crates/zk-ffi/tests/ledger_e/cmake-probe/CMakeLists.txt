# The CMake probe of tests/ledger_e_build.rs.
#
# A CMake application includes the C client with add_subdirectory(). The
# probe prints the usage requirements of the zookeeper target. Thus the
# test sees the compile definitions and the link interface that an
# application gets. The probe only configures. It
# builds nothing.
#
# Give the C client directory as ZK_C_SOURCE_DIR. The other options go to
# the CMakeLists.txt of the C client unchanged.

cmake_minimum_required(VERSION 3.15)
project(zk_ledger_e_probe C)

if(NOT ZK_C_SOURCE_DIR)
  message(FATAL_ERROR "give -DZK_C_SOURCE_DIR=<zookeeper-client-c>")
endif()
add_subdirectory("${ZK_C_SOURCE_DIR}" zookeeper)

get_target_property(zk_probe_defs zookeeper INTERFACE_COMPILE_DEFINITIONS)
get_target_property(zk_probe_libs zookeeper INTERFACE_LINK_LIBRARIES)
message(STATUS "ZKPROBE defs=${zk_probe_defs}")
message(STATUS "ZKPROBE libs=${zk_probe_libs}")
